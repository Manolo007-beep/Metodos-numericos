clear; clc;
format long;

% --- CONFIGURACIÓN ---
f = @(x) x^2 - 2*cos(2*x); % Función
a = 0.5;                   % Extremo izquierdo
b = 1.0;                   % Extremo derecho
tol = 1e-5;                % Tolerancia (opcional)
max_iter = 10;             % Límite de iteraciones
% ---------------------

fprintf('--- Método de Bisección ---\n');
fprintf('it |      a      |      b      |      c      |     f(c)    |    Error    \n');
fprintf('--------------------------------------------------------------------------\n');

for k = 1:max_iter
    c = (a + b) / 2;
    fc = f(c);
    error_aprox = abs(b - a) / 2; % Cota de error en bisección
    
    fprintf('%2d | %10.6f | %10.6f | %10.6f | %11.6f | %10.6e\n', k, a, b, c, fc, error_aprox);
    
    if abs(fc) < 1e-12 || error_aprox < tol
        break;
    end
    
    % Decisión del nuevo intervalo
    if f(a) * fc < 0
        b = c;
    else
        a = c;
    end
end
