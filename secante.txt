clear; clc;
format long;

% --- CONFIGURACIÓN ---
f = @(x) x^2 - 2*cos(2*x); 
x0 = 0.5; 
x1 = 1.0;
max_iter = 8;
% ---------------------

fprintf('--- Método de la Secante ---\n');
fprintf('it |      xi-1     |      xi       |      xi+1     |    Error abs   \n');
fprintf('----------------------------------------------------------------------\n');

for k = 1:max_iter
    fx0 = f(x0);
    fx1 = f(x1);
    
    % Fórmula de la secante
    xi_mas_1 = x1 - fx1 * (x1 - x0) / (fx1 - fx0);
    
    error_abs = abs(xi_mas_1 - x1);
    
    fprintf('%2d | %11.6f | %11.6f | %11.6f | %11.6e\n', k, x0, x1, xi_mas_1, error_abs);
    
    % Actualización de puntos
    x0 = x1;
    x1 = xi_mas_1;
    
    if error_abs < 1e-10
        break;
    end
end
